# 4.4 Vision Request Shape & Structured Output JSON Schema

Kimi-K3 requests use the OpenAI-compatible `/v1/chat/completions` contract. The model identifier must remain configurable because NVIDIA may expose the deployment under a different name, for example:

```env
NVIDIA_NIM_MODEL=moonshotai/kimi-k3
```

The canonical multimodal request is:

```json
{
  "model": "moonshotai/kimi-k3",
  "messages": [
    {
      "role": "system",
      "content": "You are Helmetsan Vision Auditor. Return only valid JSON."
    },
    {
      "role": "user",
      "content": [
        {
          "type": "text",
          "text": "Audit this helmet image for visible safety defects."
        },
        {
          "type": "image_url",
          "image_url": {
            "url": "data:image/jpeg;base64,/9j/4AAQ..."
          }
        }

  "temperature": 0.1,
  "max_tokens": 2000,
  "seed": 42,
  "response_format": {
    "type": "json_object"
  },
  "stream": false
}
```

Remote image URLs must not be passed directly from untrusted users. The WordPress provider accepts only:

1. Data URLs generated by the application.
2. Images fetched through a controlled media proxy.
3. WordPress attachment IDs converted server-side to data URLs.

`BaseProvider.php`:

```php
<?php
declare(strict_types=1);

namespace Helmetsan\Core\AI;

abstract class BaseProvider
{
    protected function jsonDecode(string $body): array
    {
        $decoded = json_decode($body, true);

        if (!is_array($decoded)) {
            throw new \RuntimeException('Provider returned invalid JSON.');
        }

        return $decoded;
    }

    protected function assertSuccess(int $status, string $body): void
    {
        if ($status >= 200 && $status < 300) {
            return;
        }

        $message = 'AI provider request failed.';

        $decoded = json_decode($body, true);
        if (is_array($decoded)) {
            $message = (string) (
                $decoded['error']['message']
                ?? $decoded['message']
                ?? $message
            );
        }

        $error = new \RuntimeException($message);
        $error->httpStatus = $status;
        throw $error;
    }

    protected function bearer(string $key): string
    {
        return 'Bearer ' . trim($key);
    }
}
```

## 5.2 `NvidiaNimProvider.php`

```php
<?php
declare(strict_types=1);

namespace Helmetsan\Core\AI\Providers;

use Helmetsan\Core\AI\BaseProvider;
use Helmetsan\Core\AI\ProviderInterface;
use WP_Error;

final class NvidiaNimProvider extends BaseProvider implements ProviderInterface
{
    private const PROVIDER_ID = 'nvidia_nim';
    private const ALLOWED_ORIGIN = 'https://integrate.api.nvidia.com';
    private const CHAT_PATH = '/v1/chat/completions';

    private string $apiKey;
    private string $model;
    private int $timeout;
    private string $endpoint;

    public function __construct(array $config = [])
    {
        $this->apiKey = (string) ($config['api_key'] ?? '');
        $this->model = (string) (
            $config['model']
            ?? getenv('NVIDIA_NIM_MODEL')
            ?: 'moonshotai/kimi-k3'
        );
        $this->timeout = max(5, min(180, (int) (
            $config['timeout']
            ?? getenv('NVIDIA_NIM_TIMEOUT')
            ?: 90
        )));

        $endpoint = (string) (
            $config['endpoint']
            ?? getenv('NVIDIA_NIM_ENDPOINT')
            ?: self::ALLOWED_ORIGIN
        );

        $this->endpoint = $this->validateEndpoint($endpoint);
    }

    public function id(): string
    {
        return self::PROVIDER_ID;
    }

    public function available(): bool
    {
        return $this->apiKey !== ''
            && hash_equals(
                self::ALLOWED_ORIGIN,
                rtrim($this->endpoint, '/')
            );
    }

    public function chat(array $request): array
    {
        $payload = $this->buildPayload($request);
        $url = $this->endpoint . self::CHAT_PATH;

        $response = wp_safe_remote_post($url, [
            'timeout' => $this->timeout,
            'headers' => [
                'Authorization' => $this->bearer($this->apiKey),
                'Accept' => 'application/json',
                'Content-Type' => 'application/json',
                'User-Agent' => 'Helmetsan-WordPress/' . HELMETSAN_VERSION,
            ],
            'body' => wp_json_encode($payload),
            'redirection' => 0,
            'reject_unsafe_urls' => true,
        ]);

        if ($response instanceof WP_Error) {
            throw new \RuntimeException($response->get_error_message());
        }

        $status = (int) wp_remote_retrieve_response_code($response);
        $body = (string) wp_remote_retrieve_body($response);

        $this->assertSuccess($status, $body);

        $decoded = $this->jsonDecode($body);

        return [
            'provider' => $this->id(),
            'model' => $decoded['model'] ?? $this->model,
            'id' => $decoded['id'] ?? null,
            'content' => $this->extractContent($decoded),
            'raw' => $decoded,
            'usage' => $decoded['usage'] ?? [],
        ];
    }

    public function stream(array $request, callable $onEvent): void
    {
        /*
         * WordPress HTTP does not provide a consistently portable streaming
         * interface across all supported PHP transports. The production
         * implementation uses the Mission Control Node gateway for SSE.
         *
         * This method intentionally fails closed rather than pretending that
         * a non-streaming response is a stream.
         */
        throw new \RuntimeException(
            'NVIDIA NIM streaming must be consumed through Mission Control.'
        );
    }

    /**
     * @param array<string,mixed> $request
     * @return array<string,mixed>
     */
    private function buildPayload(array $request): array
    {
        $messages = $request['messages'] ?? [];

        if (!is_array($messages) || $messages === []) {
            throw new \InvalidArgumentException('messages is required.');
        }

        $payload = [
            'model' => (string) ($request['model'] ?? $this->model),
            'messages' => $this->sanitizeMessages($messages),
            'stream' => false,
        ];

        $allowedScalarOptions = [
            'temperature',
            'top_p',
            'max_tokens',
            'seed',
            'frequency_penalty',
            'presence_penalty',
        ];

        foreach ($allowedScalarOptions as $key) {
            if (array_key_exists($key, $request)) {
                $payload[$key] = $this->validatedScalarOption(
                    $key,
                    $request[$key]
                );
            }
        }

        if (isset($request['reasoning_effort'])) {
            $effort = (string) $request['reasoning_effort'];

            if (!in_array($effort, ['low', 'medium', 'high'], true)) {
                throw new \InvalidArgumentException(
                    'reasoning_effort must be low, medium, or high.'
                );
            }

            /*
             * NIM-compatible deployments may implement this as a native field
             * or as an extra_body field. Keep the mapping configurable.
             */
            $payload['reasoning_effort'] = $effort;
        }

        if (!empty($request['json_object'])) {
            $payload['response_format'] = [
                'type' => 'json_object',
            ];
        }

        if (isset($request['response_format'])) {
            $payload['response_format'] = $request['response_format'];
        }

        return $payload;
    }

    /**
     * @param array<int,mixed> $messages
     * @return array<int,array<string,mixed>>
     */
    private function sanitizeMessages(array $messages): array
    {
        $result = [];

        foreach ($messages as $message) {
            if (!is_array($message)) {
                throw new \InvalidArgumentException('Invalid message.');
            }

            $role = (string) ($message['role'] ?? '');

            if (!in_array($role, ['system', 'user', 'assistant'], true)) {
                throw new \InvalidArgumentException('Invalid message role.');
            }

            $content = $message['content'] ?? '';

            if (is_string($content)) {
                $result[] = [
                    'role' => $role,
                    'content' => $content,
                ];
                continue;
            }

            if (!is_array($content)) {
                throw new \InvalidArgumentException('Invalid message content.');
            }

            $parts = [];

            foreach ($content as $part) {
                if (!is_array($part)) {
                    continue;
                }

                $type = $part['type'] ?? '';

                if ($type === 'text') {
                    $parts[] = [
                        'type' => 'text',
                        'text' => (string) ($part['text'] ?? ''),
                    ];
                } elseif ($type === 'image_url') {
                    $url = (string) ($part['image_url']['url'] ?? '');

                    if (!$this->isAllowedImageSource($url)) {
                        throw new \InvalidArgumentException(
                            'Image source must be a data URL.'
                        );
                    }

                    $parts[] = [
                        'type' => 'image_url',
                        'image_url' => ['url' => $url],
                    ];
                }
            }

            if ($parts === []) {
                throw new \InvalidArgumentException(
                    'Multimodal content contains no valid parts.'
                );
            }

            $result[] = [
                'role' => $role,
                'content' => $parts,
            ];
        }

        return $result;
    }

    private function isAllowedImageSource(string $url): bool
    {
        return (bool) preg_match(
            '#^data:image/(jpeg|jpg|png|webp);base64,[A-Za-z0-9+/=]+$#',
            $url
        );
    }

    private function validateEndpoint(string $endpoint): string
    {
        $parts = wp_parse_url($endpoint);

        if (
            !$parts
            || ($parts['scheme'] ?? '') !== 'https'
            || ($parts['host'] ?? '') !== 'integrate.api.nvidia.com'
            || isset($parts['user'])
            || isset($parts['pass'])
        ) {
            throw new \InvalidArgumentException(
                'NVIDIA endpoint is not on the approved allowlist.'
            );
        }

        return 'https://integrate.api.nvidia.com';
    }

    private function validatedScalarOption(string $key, mixed $value): mixed
    {
        return match ($key) {
            'max_tokens' => max(1, min(1000000, (int) $value)),
            'seed' => (int) $value,
            'temperature', 'top_p',
            'frequency_penalty', 'presence_penalty' => (float) $value,
            default => $value,
        };
    }

    private function extractContent(array $response): string
    {
        $content = $response['choices'][0]['message']['content'] ?? '';

        if (is_string($content)) {
            return $content;
        }

        return wp_json_encode($content) ?: '';
    }
}
```

### Important SSRF constraints

The provider must not accept a user-supplied provider URL. The endpoint is:

- HTTPS only.
- Host exactly `integrate.api.nvidia.com`.
- No credentials.
- No redirects.
- No arbitrary ports.
- No user-controlled URL path.

Image URLs are converted into data URLs before reaching the provider.

`ProviderRegistry.php`:

```php
<?php
declare(strict_types=1);

namespace Helmetsan\Core\AI;

use Helmetsan\Core\AI\Providers\NvidiaNimProvider;

final class ProviderRegistry
{
    /** @var array<string,ProviderInterface> */
    private array $providers = [];

    public function __construct(Config $config)
    {
        $this->providers['nvidia_nim'] = new NvidiaNimProvider([
            'api_key' => $config->nvidiaApiKey(),
            'model' => $config->nvidiaModel(),
            'endpoint' => $config->nvidiaEndpoint(),
            'timeout' => $config->nvidiaTimeout(),
        ]);
    }

    public function get(string $id): ProviderInterface
    {
        if (!isset($this->providers[$id])) {
            throw new \InvalidArgumentException(
                "Unknown AI provider: {$id}"
            );
        }

        return $this->providers[$id];
    }

    /**
     * @return
